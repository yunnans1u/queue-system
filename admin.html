// 打开统计弹窗 window.openStatsModal = function() { const modal = document.getElementById('statsModal'); modal.classList.add('active'); loadStatsData(); } // 关闭统计弹窗 window.closeStatsModal = function() { const modal = document.getElementById('statsModal'); modal.classList.remove('active'); } // 点击弹窗外部关闭 window.onclick = function(event) { const modal = document.getElementById('statsModal'); if (event.target === modal) { closeStatsModal(); } } // 加载统计数据 function loadStatsData() { const content = document.getElementById('statsContent'); const stats = getQueueStatistics(); if (Object.keys(stats).length === 0) { content.innerHTML = '
暂无数据，等待顾客完成订单后自动记录 📝
'; return; } let html = ''; // 按日期排序（最新的在前面） const sortedDates = Object.keys(stats).sort().reverse(); sortedDates.forEach(date => { const dayData = stats[date]; html += `
${date}
总订单数
${dayData.totalOrders}
平均等待
${dayData.averageWait}分钟
⏰ 高峰时段
${dayData.peakHourAvg}分钟
12-2pm, 5:30-9pm
🌿 非高峰
${dayData.offPeakAvg}分钟
${dayData.data.map(order => ` `).join('')}
排队号 预估 实际 时段
${order.queueNumber} ${order.estimatedWaitTime}分钟 ${order.actualWaitTime}分钟 ${order.peakHour ? '⏰高峰' : '🌿平峰'}
`; }); // 添加导出按钮 html += `
`; content.innerHTML = html; } // 切换详细订单显示 window.toggleDetail = function(date) { const detail = document.getElementById(`detail-${date}`); detail.classList.toggle('show'); } // 获取统计数据（从localStorage读取） window.getQueueStatistics = function() { const stats = {}; const today = new Date(); for (let i = 0; i < 7; i++) { const date = new Date(today); date.setDate(date.getDate() - i); const dateStr = date.toISOString().split('T')[0]; const dayData = JSON.parse(localStorage.getItem(`queue_data_${dateStr}`) || '[]'); if (dayData.length > 0) { const peakWaitTimes = dayData.filter(d => d.peakHour).map(d => d.actualWaitTime); const offPeakWaitTimes = dayData.filter(d => !d.peakHour).map(d => d.actualWaitTime); stats[dateStr] = { totalOrders: dayData.length, averageWait: Math.round(dayData.reduce((sum, d) => sum + d.actualWaitTime, 0) / dayData.length), peakHourAvg: peakWaitTimes.length ? Math.round(peakWaitTimes.reduce((sum, t) => sum + t, 0) / peakWaitTimes.length) : 0, offPeakAvg: offPeakWaitTimes.length ? Math.round(offPeakWaitTimes.reduce((sum, t) => sum + t, 0) / offPeakWaitTimes.length) : 0, data: dayData }; } } return stats; } // 导出为CSV window.exportStats = function() { const stats = getQueueStatistics(); let csv = '日期,排队号,预估时间,实际时间,时段\n'; Object.keys(stats).forEach(date => { stats[date].data.forEach(order => { csv += `${date},${order.queueNumber},${order.estimatedWaitTime},${order.actualWaitTime},${order.peakHour ? '高峰' : '平峰'}\n`; }); }); const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' }); const link = document.createElement('a'); const url = URL.createObjectURL(blob); link.setAttribute('href', url); link.setAttribute('download', `queue_stats_${new Date().toISOString().split('T')[0]}.csv`); link.style.visibility = 'hidden'; document.body.appendChild(link); link.click(); document.body.removeChild(link); } // 自动清理旧数据（保留7天） function cleanOldStats() { const today = new Date(); for (let i = 7; i <= 30; i++) { const date = new Date(today); date.setDate(date.getDate() - i); const dateStr = date.toISOString().split('T')[0]; localStorage.removeItem(`queue_data_${dateStr}`); } } // 每天执行一次清理 setInterval(cleanOldStats, 24 * 60 * 60 * 1000);
